<script>
    let chart;
    const ctx = document.getElementById('graphCanvas').getContext('2d');

    // Setup input listeners to automatically update graph and calculations
    const inputs = ['functionInput', 'lowerBound', 'upperBound', 'rectangles', 'sumType'];
    inputs.forEach(id => {
        document.getElementById(id).addEventListener('input', calculateAndPlot);
    });

    // Helper to evaluate f(x) safely using math.js
    function evaluateFunction(expr, xVal) {
        try {
            return math.evaluate(expr, { x: xVal });
        } catch (e) {
            return null;
        }
    }

    function calculateAndPlot() {
        const expr = document.getElementById('functionInput').value;
        const a = parseFloat(document.getElementById('lowerBound').value);
        const b = parseFloat(document.getElementById('upperBound').value);
        const n = parseInt(document.getElementById('rectangles').value) || 10;
        const sumType = document.getElementById('sumType').value;
        const errorMsg = document.getElementById('errorMsg');

        // Reset error message
        errorMsg.style.display = 'none';

        // Basic validation
        if (isNaN(a) || !isFinite(a) || isNaN(b) || !isFinite(b) || a >= b || n <= 0) {
            document.getElementById('resultValue').innerText = "---";
            return;
        }

        // Test math string validity
        if (evaluateFunction(expr, a) === null) {
            errorMsg.style.display = 'block';
            document.getElementById('resultValue').innerText = "---";
            return;
        }

        // 1. Calculate Riemann Sum
        const dx = (b - a) / n;
        let totalArea = 0;
        let rectBars = [];

        for (let i = 0; i < n; i++) {
            let xEval;
            let xLeft = a + i * dx;
            let xRight = a + (i + 1) * dx;

            if (sumType === 'left') {
                xEval = xLeft;
            } else if (sumType === 'right') {
                xEval = xRight;
            } else {
                xEval = xLeft + dx / 2; // Midpoint
            }

            let yVal = evaluateFunction(expr, xEval);
            if (typeof yVal !== 'number' || isNaN(yVal)) yVal = 0;
            
            totalArea += yVal * dx;

            // Generate coordinates to render standard bar rectangles manually
            rectBars.push({ xLeft, xRight, yVal });
        }

        document.getElementById('resultValue').innerText = totalArea.toFixed(5);

        // 2. Generate Smooth Curve Data points
        const curvePoints = [];
        // Add padding to graph view outside bounds [a, b]
        const padding = (b - a) * 0.25 || 1;
        const graphMinX = a - padding;
        const graphMaxX = b + padding;
        const steps = 150;
        const stepSize = (graphMaxX - graphMinX) / steps;

        for (let i = 0; i <= steps; i++) {
            let currX = graphMinX + i * stepSize;
            let currY = evaluateFunction(expr, currX);
            if (typeof currY === 'number' && !isNaN(currY)) {
                curvePoints.push({ x: currX, y: currY });
            }
        }

        // 3. Build Chart Data structures
        // Line data for the function
        const datasets = [{
            label: `f(x) = ${expr}`,
            data: curvePoints,
            borderColor: '#2563eb',
            borderWidth: 3,
            fill: false,
            pointRadius: 0,
            tension: 0.1,
            order: 1
        }];

        // Dynamic individual dataset for each Riemann rectangle bar
        rectBars.forEach((bar, idx) => {
            datasets.push({
                label: idx === 0 ? 'Riemann Sum Area' : '', // Label only first for clean legend
                data: [
                    { x: bar.xLeft, y: 0 },
                    { x: bar.xLeft, y: bar.yVal },
                    { x: bar.xRight, y: bar.yVal },
                    { x: bar.xRight, y: 0 }
                ],
                borderColor: 'rgba(59, 130, 246, 0.6)',
                backgroundColor: 'rgba(59, 130, 246, 0.25)',
                borderWidth: 1,
                fill: true,
                pointRadius: 0,
                showLine: true,
                order: 2
            });
        });

        // 4. Render/Update Chart
        if (chart) {
            chart.data.datasets = datasets;
            chart.options.scales.x.min = graphMinX;
            chart.options.scales.x.max = graphMaxX;
            chart.update('none'); // Update without full recalculation lag
        } else {
            chart = new Chart(ctx, {
                type: 'scatter',
                data: { datasets: datasets },
                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    plugins: {
                        legend: {
                            labels: {
                                filter: function(item, chartData) {
                                    // Hide repetitive legend items for rectangles
                                    return item.text !== '';
                                }
                            }
                        }
                    },
                    scales: {
                        x: {
                            type: 'linear',
                            position: 'bottom',
                            min: graphMinX,
                            max: graphMaxX,
                            grid: { color: '#e2e8f0' }
                        },
                        y: {
                            type: 'linear',
                            grid: { color: '#e2e8f0' }
                        }
                    },
                    animation: false
                }
            });
        }
    }

    // Run layout on start
    calculateAndPlot();
</script>
